# Brezel's tests (README.md "Tests", docs/Framework_Guide.md section 14).
#
# One binary per area, each built twice: as C++20 (the standard of the project that uses Brezel) and as C++23 (the
# standard CMakeLists.txt declares). Every binary is registered with CTest as brezel_<area>_cxx<standard>. The test
# framework is tests/BrezelTest.hpp: no dependency beyond Brezel's own (EnTT, spdlog, tinyxml2) and the threads library.
#
# Standalone: cmake -S deps/Brezel -B Build-brezel -G Ninja -DBREZEL_BUILD_TESTS=ON, then ninja and ctest there.
# Inside a parent project: the parent sets BREZEL_BUILD_TESTS ON before add_subdirectory, and defines the spdlog and
# tinyxml2 targets and EnTT's include path for the Brezel target, as it does for its own code (README.md).

find_package(Threads REQUIRED)

set(BREZEL_TEST_INCLUDE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../include)

# What every test links: Brezel's headers and their dependencies, the threads library
add_library(brezel_test_support INTERFACE)
target_include_directories(brezel_test_support INTERFACE ${CMAKE_CURRENT_SOURCE_DIR} ${BREZEL_TEST_INCLUDE_DIR})
if(BREZEL_STANDALONE)
    # Standalone: EnTT, spdlog and tinyxml2 from extern/ (the Brezel target is the testbed when BREZEL_BUILD_TESTBED)
    target_link_libraries(brezel_test_support INTERFACE EnTT::EnTT spdlog::spdlog tinyxml2::tinyxml2)
else()
    # Subdirectory: the parent's Brezel INTERFACE target carries EnTT's include path, spdlog and tinyxml2
    target_link_libraries(brezel_test_support INTERFACE Brezel)
endif()
target_link_libraries(brezel_test_support INTERFACE Threads::Threads)

set(BREZEL_TEST_AREAS
    core            # Project, Entity, UUID, identity, hierarchy, references, the ECS wall, the component registry
)

foreach(area IN LISTS BREZEL_TEST_AREAS)
    foreach(standard 20 23)
        set(name brezel_${area}_cxx${standard})
        add_executable(${name} test_${area}.cpp)
        target_link_libraries(${name} PRIVATE brezel_test_support)
        set_target_properties(${name} PROPERTIES
            CXX_STANDARD ${standard}
            CXX_STANDARD_REQUIRED ON
            CXX_EXTENSIONS OFF)
        add_test(NAME ${name} COMMAND ${name})
        # Every thread in a test runs under BrezelTest::Watchdog (a few seconds); this is the outer bound
        set_tests_properties(${name} PROPERTIES TIMEOUT 300 LABELS brezel)
    endforeach()
endforeach()

# The header index of the guide (docs/Framework_Guide.md section 13): every header compiles in a translation unit of
# its own, after the header its "First" column names, and all of them link into one program (no duplicate symbol).
# InteractiveConsole.hpp is left out: it needs GNU readline, which is not a dependency of the tests.
set(BREZEL_HEADERS_ALONE
    Command/Command.hpp Command/CommandStack.hpp
    Concurrency/AnyMutation.hpp Concurrency/MutationQueue.hpp Concurrency/PeriodicLoop.hpp
    Concurrency/QueueComponent.hpp Concurrency/RingBuffer.hpp Concurrency/StallWatchdog.hpp
    Concurrency/StandardMutations.hpp Concurrency/StreamComponent.hpp Concurrency/ThreadUtils.hpp
    Concurrency/TriggeredCapture.hpp Concurrency/TripleBuffer.hpp
    Console/SearchVisitor.hpp
    Core/ComponentRegistry.hpp Core/Dispatcher.hpp Core/EcsWall.hpp Core/Entity.hpp Core/EntityReference.hpp
    Core/Project.hpp Core/UUID.hpp Core/UUIDProvider.hpp
    Foundation/MonotonicRawClock.hpp Foundation/Signal.hpp Foundation/StringID.hpp
    Math/CircleFit.hpp Math/LeastSquares.hpp Math/LineFit.hpp
    Reflection/Visitor.hpp
    Serialization/BinaryVisitor.hpp Serialization/DeserializationValidation.hpp Serialization/XmlCommon.hpp
    Time/RemoteClockEstimator.hpp
)
# header|first[;first...] for the headers that need another one first
set(BREZEL_HEADERS_WITH_FIRST
    "App/Application.hpp|Core/ComponentRegistry.hpp"
    "Console/Console.hpp|Core/ComponentRegistry.hpp"
    "Core/CoreComponents.hpp|Core/Entity.hpp"
    "Serialization/ConsoleVisitor.hpp|Core/ComponentRegistry.hpp,Core/EntityReference.hpp"
    "Serialization/EntityReferenceLinker.hpp|Core/ComponentRegistry.hpp"
    "Serialization/XmlDeserializer.hpp|Core/ComponentRegistry.hpp"
    "Serialization/XmlSerializer.hpp|Core/ComponentRegistry.hpp"
)

set(header_sources "")
foreach(header IN LISTS BREZEL_HEADERS_ALONE)
    string(MAKE_C_IDENTIFIER ${header} id)
    set(file ${CMAKE_CURRENT_BINARY_DIR}/headers/${id}.cpp)
    file(CONFIGURE OUTPUT ${file} CONTENT "// Generated by tests/CMakeLists.txt: Brezel/${header} compiles alone\n#include \"Brezel/${header}\"\n#include \"Brezel/${header}\"\n")
    list(APPEND header_sources ${file})
endforeach()
foreach(entry IN LISTS BREZEL_HEADERS_WITH_FIRST)
    string(REPLACE "|" ";" parts "${entry}")
    list(GET parts 0 header)
    list(GET parts 1 firsts)
    string(REPLACE "," ";" firsts "${firsts}")
    set(content "// Generated by tests/CMakeLists.txt: Brezel/${header} compiles after the header(s) it needs first\n")
    foreach(first IN LISTS firsts)
        string(APPEND content "#include \"Brezel/${first}\"\n")
    endforeach()
    string(APPEND content "#include \"Brezel/${header}\"\n")
    string(MAKE_C_IDENTIFIER ${header} id)
    set(file ${CMAKE_CURRENT_BINARY_DIR}/headers/${id}.cpp)
    file(CONFIGURE OUTPUT ${file} CONTENT "${content}")
    list(APPEND header_sources ${file})
endforeach()

add_executable(brezel_headers_cxx20 test_headers.cpp ${header_sources})
target_link_libraries(brezel_headers_cxx20 PRIVATE brezel_test_support)
set_target_properties(brezel_headers_cxx20 PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
add_test(NAME brezel_headers_cxx20 COMMAND brezel_headers_cxx20)
set_tests_properties(brezel_headers_cxx20 PROPERTIES TIMEOUT 60 LABELS brezel)
